const HOURS = {
  1: [[9, 0, 12, 0], [14, 0, 17, 0]],
  2: [[9, 0, 12, 0], [14, 0, 17, 0]],

  6: [[9, 0, 12, 0]],
  0: []
};
const SLOT_MIN = 30;
const DAYS_AHEAD = 30;
const SHEET_API_URL = "https://script.google.com/macros/s/AKfycbxJKeinBwanTHIohLTrPgD5Dq1vRjgGT8yY3FPk0W88_lzG7T5apu_aexNBcazh6Hv-EQ/exec";

function fmtDate(d) {
  const y = d.getFullYear();
  const m = String(d.getMonth() + 1).padStart(2, '0');
  const day = String(d.getDate()).padStart(2, '0');
  return `${y}-${m}-${day}`;
}

function parseLocalDate(dateStr) {
  const [y, m, d] = dateStr.split('-').map(Number);
  return new Date(y, m - 1, d);
}

function fmtLabel(dateStr) {
  if (!dateStr) return '';
  const d = parseLocalDate(dateStr);
  return d.toLocaleDateString('pt-BR', { weekday: 'long', day: '2-digit', month: 'long' });
}

function makeBookingCode(childName, dateStr, time) {
  const namePart = childName
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '')
    .replace(/[^a-zA-Z]/g, '')
    .toUpperCase();
  const datePart = dateStr.replace(/-/g, '');
  const timePart = time.replace(':', '');
  return `${namePart}-${datePart}-${timePart}`;
}

function normalizeTime(raw) {
  if (raw === null || raw === undefined) return '';
  if (typeof raw === 'string') {
    const m = raw.match(/^(\d{1,2}):(\d{2})/);
    if (m) return m[1].padStart(2, '0') + ':' + m[2];
  }
  const d = new Date(raw);
  if (!isNaN(d.getTime())) {
    let mins = d.getHours() * 60 + d.getMinutes();
    mins = Math.round(mins / SLOT_MIN) * SLOT_MIN;
    mins = ((mins % 1440) + 1440) % 1440;
    return String(Math.floor(mins / 60)).padStart(2, '0') + ':' + String(mins % 60).padStart(2, '0');
  }
  return String(raw);
}

function getSlotsForDate(dateStr, bookingsList) {
  if (!dateStr) return [];
  const d = parseLocalDate(dateStr);
  const periods = HOURS[d.getDay()] || [];
  let slots = [];
  periods.forEach(([sh, sm, eh, em]) => {
    let start = sh * 60 + sm, end = eh * 60 + em;
    for (let t = start; t < end; t += SLOT_MIN) {
      const hh = String(Math.floor(t / 60)).padStart(2, '0');
      const mm = String(t % 60).padStart(2, '0');
      slots.push(hh + ":" + mm);
    }
  });
  const taken = (bookingsList[dateStr] || []).map(b => b.time);
  const now = new Date();
  return slots.filter(s => {
    if (taken.includes(s)) return false;
    if (fmtDate(now) === dateStr) {
      const [h, m] = s.split(":").map(Number);
      if (h * 60 + m <= now.getHours() * 60 + now.getMinutes()) return false;
    }
    return true;
  });
}

// API HELPERS
async function apiGet() {
  return new Promise((resolve, reject) => {
    const cbName = 'sheetsCb_' + Date.now();
    window[cbName] = function (data) {
      resolve(data);
      delete window[cbName];
      script.remove();
    };
    const script = document.createElement('script');
    const sep = SHEET_API_URL.includes('?') ? '&' : '?';
    script.src = SHEET_API_URL + sep + 'callback=' + cbName;
    script.onerror = function () {
      delete window[cbName];
      script.remove();
      reject(new Error('Falha ao carregar dados.'));
    };
    document.body.appendChild(script);
  });
}

async function apiPost(payload) {
  await fetch(SHEET_API_URL, { method: 'POST', mode: 'no-cors', body: JSON.stringify(payload) });
}

async function checkPasswordRemote(password) {
  return new Promise((resolve) => {
    const cbName = 'passCb_' + Date.now();
    window[cbName] = function (data) {
      resolve(data && data.ok === true);
      delete window[cbName];
      script.remove();
    };
    const script = document.createElement('script');
    const sep = SHEET_API_URL.includes('?') ? '&' : '?';
    script.src = SHEET_API_URL + sep + 'action=verifyPassword&password=' + encodeURIComponent(password) + '&callback=' + cbName;
    script.onerror = function () {
      delete window[cbName];
      script.remove();
      resolve(false);
    };
    document.body.appendChild(script);
  });
}

export default function SorrisoMagicoBooking() {
  const [step, setStep] = useState(1);
  const [bookings, setBookings] = useState({});
  const [blockedDays, setBlockedDays] = useState([]);
  
  const [selectedDate, setSelectedDate] = useState(null);
  const [selectedTime, setSelectedTime] = useState(null);
  const [selectedMission, setSelectedMission] = useState('');
  const [formData, setFormData] = useState({ childName: '', childAge: '', parentName: '', phone: '' });

  const [step3Error, setStep3Error] = useState(false);
  const [createdBookingCode, setCreatedBookingCode] = useState('');
  const [createdBookingDetails, setCreatedBookingDetails] = useState(null);
  const [isSubmitting, setIsSubmitting] = useState(false);

  const [isAdminOpen, setIsAdminOpen] = useState(false);
  const [isAdminAuthenticated, setIsAdminAuthenticated] = useState(false);
  const [adminPass, setAdminPass] = useState('');
  const [adminError, setAdminError] = useState(false);
  const [adminLoading, setAdminLoading] = useState(false);
  const [blockDayInput, setBlockDayInput] = useState('');

  // CARREGAR DADOS
  const loadData = useCallback(async () => {
    if (USING_SHEETS) {
      try {
        const data = await apiGet();
        const loadedBookings = data.bookings || {};
        Object.keys(loadedBookings).forEach((date) => {
          loadedBookings[date] = loadedBookings[date].map((b) => ({ ...b, time: normalizeTime(b.time) }));
        });
        setBookings(loadedBookings);
        setBlockedDays([...new Set(data.blockedDays || [])]);
        return;
      } catch (e) {
        console.error('Falha na API, usando dados locais.', e);
      }
    }
    
    let localB = {}, localD = [];
    try { localB = JSON.parse(localStorage.getItem('bookings-all') || '{}'); } catch (e) {}
    try { localD = JSON.parse(localStorage.getItem('blocked-days') || '[]'); } catch (e) {}

    setBookings(localB);
    setBlockedDays([...new Set(localD)]);
  }, []);

  useEffect(() => { loadData(); }, [loadData]);

  const saveLocalFallback = (newBookings, newBlocked) => {
    localStorage.setItem('bookings-all', JSON.stringify(newBookings));
    localStorage.setItem('blocked-days', JSON.stringify([...new Set(newBlocked)]));
  };

  // MEMOIZAÇÃO: CÁLCULO DOS DIAS (SÓ RECOMPUTA SE A DATA BLOQUEADA OU AGENDAMENTOS MUSAREM)
  const availableDays = useMemo(() => {
    const items = [];
    const today = new Date();
    let shown = 0, i = 0;

    while (shown < DAYS_AHEAD) {
      const d = new Date(today.getFullYear(), today.getMonth(), today.getDate() + i);
      i++;
      const dateStr = fmtDate(d);
      const isClosed = (HOURS[d.getDay()] || []).length === 0;
      const isBlocked = blockedDays.includes(dateStr);
      const slots = getSlotsForDate(dateStr, bookings);
      const hasSlots = !isClosed && !isBlocked && slots.length > 0;

      if (isClosed) continue;
      shown++;

      items.push({
        dateStr,
        dayNum: d.getDate(),
        dayName: DAY_NAMES[d.getDay()],
        hasSlots
      });
    }
    return items;
  }, [blockedDays, bookings]);

  // MEMOIZAÇÃO: HORÁRIOS DA DATA SELECIONADA
  const selectedDaySlots = useMemo(() => {
    return getSlotsForDate(selectedDate, bookings);
  }, [selectedDate, bookings]);

  // NAVEGAÇÃO
  const handleNextToStep2 = async () => {
    await loadData();
    setSelectedTime(null);
    setStep(2);
  };

  const handleNextToStep4 = () => {
    const { childName, childAge, parentName, phone } = formData;
    const age = parseInt(childAge, 10);
    const isMinor = !isNaN(age) && age < 18;
    const isParentValid = isMinor ? parentName.trim().length > 0 : true;

    if (!selectedMission || !childName.trim() || childAge === '' || isNaN(age) || !phone.trim() || !isParentValid) {
      setStep3Error(true);
      return;
    }
    setStep3Error(false);
    setStep(4);
  };

  const handleConfirmBooking = async () => {
    setIsSubmitting(true);
    await loadData();
    
    if (!selectedDaySlots.includes(selectedTime)) {
      setIsSubmitting(false);
      alert("O horário selecionado não está mais disponível. Escolha outro horário.");
      setStep(2);
      return;
    }

    const ageVal = parseInt(formData.childAge, 10);
    const isMinor = ageVal < 18;

    const entry = {
      time: selectedTime,
      mission: selectedMission,
      childName: formData.childName.trim(),
      childAge: formData.childAge,
      parentName: isMinor ? formData.parentName.trim() : 'Próprio(a)',
      phone: formData.phone.trim(),
      createdAt: new Date().toISOString()
    };

    try {
      if (USING_SHEETS) {
        await apiPost({ action: 'newBooking', date: selectedDate, ...entry });
      } else {
        const updated = { ...bookings, [selectedDate]: [...(bookings[selectedDate] || []), entry] };
        setBookings(updated);
        saveLocalFallback(updated, blockedDays);
      }

      setCreatedBookingCode(makeBookingCode(entry.childName, selectedDate, selectedTime));
      setCreatedBookingDetails(entry);
      setStep(5);
    } catch (e) {
      alert("Erro ao registrar agendamento. Tente novamente.");
    } finally {
      setIsSubmitting(false);
    }
  };

  const handleResetBooking = async () => {
    setSelectedDate(null);
    setSelectedTime(null);
    setSelectedMission('');
    setFormData({ childName: '', childAge: '', parentName: '', phone: '' });
    setStep3Error(false);
    await loadData();
    setStep(1);
  };

  // ADMIN ACTIONS
  const handleAdminEnter = async () => {
    if (!adminPass.trim()) return;
    setAdminLoading(true);
    const isValid = await checkPasswordRemote(adminPass.trim());
    setAdminLoading(false);

    if (isValid) {
      setAdminError(false);
      await loadData();
      setIsAdminAuthenticated(true);
    } else {
      setAdminError(true);
    }
  };

  const handleCancelBooking = async (date, idx) => {
    if (!window.confirm("Cancelar agendamento?")) return;

    const target = bookings[date][idx];
    const newDateBookings = [...bookings[date]];
    newDateBookings.splice(idx, 1);

    const updated = { ...bookings };
    if (newDateBookings.length === 0) delete updated[date];
    else updated[date] = newDateBookings;

    setBookings(updated);
    saveLocalFallback(updated, blockedDays);

    if (USING_SHEETS) {
      try { await apiPost({ action: 'cancelBooking', date, time: target.time, childName: target.childName }); } catch (e) {}
    }
  };

  const handleBlockDay = async () => {
    if (!blockDayInput || blockedDays.includes(blockDayInput)) return;

    const updated = [...new Set([...blockedDays, blockDayInput])];
    setBlockedDays(updated);
    saveLocalFallback(bookings, updated);
    setBlockDayInput('');

    if (USING_SHEETS) {
      try { await apiPost({ action: 'blockDay', date: blockDayInput }); } catch (e) {}
    }
  };

  const handleUnblockDay = async (date) => {
    const updated = blockedDays.filter(d => d !== date);
    setBlockedDays(updated);
    saveLocalFallback(bookings, updated);

    if (USING_SHEETS) {
      try { await apiPost({ action: 'unblockDay', date }); } catch (e) {}
    }
  };

  const showParentField = useMemo(() => {
    const ageNum = parseInt(formData.childAge, 10);
    return !isNaN(ageNum) && ageNum < 18;
  }, [formData.childAge]);

  const adminRows = useMemo(() => {
    const rows = [];
    Object.keys(bookings).sort().forEach(date => {
      (bookings[date] || []).forEach((b, idx) => rows.push({ date, idx, ...b }));
    });
    return rows;
  }, [bookings]);

  return (
    <>
      <style>{STATIC_STYLES}</style>

      {/* NAV */}
      <nav>
        <div className="wrap">
          <div className="brand">
            <svg className="brand-mark" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2"><path d="M12 22s8-4 8-10V5l-8-3-8 3v7c0 6 8 10 8 10z"/><path d="m9 12 2 2 4-4"/></svg>
            Sorriso Mágico
          </div>
          <a href="#agendar" className="btn btn-primary btn-sm">Agendar Consulta</a>
        </div>
      </nav>

      {/* HERO */}
      <section className="hero">
        <div className="wrap container">
          <div>
            <div className="hero-badge">Odontologia Especializada</div>
            <h1>Atendimento odontológico moderno, preciso e seguro.</h1>
            <p className="lead">Agende sua consulta ou a de seu dependente com praticidade. Escolha o melhor dia e horário em poucos passos.</p>
            <div className="actions">
              <a href="#agendar" className="btn btn-primary">Ver Horários Disponíveis</a>
              <a href="#como-funciona" className="btn btn-ghost">Como Funciona</a>
            </div>
          </div>
          
          <div className="hero-card">
            <div className="hero-card-item">
              <div className="hero-icon"><svg width="20" height="20" fill="none" stroke="currentColor" strokeWidth="2" viewBox="0 0 24 24"><path d="M8 7V3m8 4V3m-9 8h10M5 21h14a2 2 0 002-2V7a2 2 0 00-2-2H5a2 2 0 00-2 2v12a2 2 0 002 2z"/></svg></div>
              <div><b style={{ fontSize: '14px', display: 'block' }}>Agendamento em Tempo Real</b><span style={{ fontSize: '13px', color: 'var(--ink-soft)' }}>Horários atualizados instantaneamente.</span></div>
            </div>
            <div className="hero-card-item">
              <div className="hero-icon"><svg width="20" height="20" fill="none" stroke="currentColor" strokeWidth="2" viewBox="0 0 24 24"><path d="M9 12l2 2 4-4m5.618-4.016A11.955 11.955 0 0112 2.944a11.955 11.955 0 01-8.618 3.04A12.02 12.02 0 003 9c0 5.591 3.824 10.29 9 11.622 5.176-1.332 9-6.03 9-11.622 0-1.042-.133-2.052-.382-3.016z"/></svg></div>
              <div><b style={{ fontSize: '14px', display: 'block' }}>Confirmação Garantida</b><span style={{ fontSize: '13px', color: 'var(--ink-soft)' }}>Receba o código de validação da sua vaga.</span></div>
            </div>
          </div>
        </div>
      </section>

      {/* HOW IT WORKS */}
      <section className="how" id="como-funciona">
        <div className="container">
          <h2 style={{ fontSize: '22px' }}>Passo a passo para o agendamento</h2>
          <div className="how-grid">
            <div className="how-card"><div className="how-num">1</div><h3>Data do Atendimento</h3><p>Selecione um dos dias disponíveis em nosso calendário clínico.</p></div>
            <div className="how-card"><div className="how-num">2</div><h3>Horário Preferencial</h3><p>Escolha o horário livre que melhor se adapta à sua rotina.</p></div>
            <div className="how-card"><div className="how-num">3</div><h3>Identificação</h3><p>Informe os dados do paciente e confirme sua reserva.</p></div>
          </div>
        </div>
      </section>

      {/* BOOKING PANEL */}
      <section className="booking" id="agendar">
        <div className="container">
          <div className="panel">

            {/* TRAIL */}
            <div className="trail">
              {["Data", "Horário", "Dados", "Revisão", "Concluído"].map((label, idx) => {
                const n = idx + 1;
                const isDone = n < step;
                const isActive = n === step;
                return (
                  <div key={label} className={`trail-step ${isDone ? 'done' : isActive ? 'active' : ''}`}>
                    <div className="trail-dot">{isDone ? "✓" : n}</div>
                    <div className="trail-label">{label}</div>
                  </div>
                );
              })}
            </div>

            <form onSubmit={(e) => e.preventDefault()}>

              {/* STEP 1: DIA */}
              <div className={`step-content ${step === 1 ? 'active' : ''}`}>
                <h2 style={{ fontSize: '18px', marginBottom: '4px' }}>Selecione a Data</h2>
                <p style={{ color: 'var(--ink-soft)', fontSize: '14px', margin: '0 0 20px' }}>Dias sem disponibilidade permanecem desabilitados.</p>
                
                <div className="day-grid">
                  {availableDays.map((item) => (
                    <div
                      key={item.dateStr}
                      className={`pick-card ${!item.hasSlots ? 'disabled' : ''} ${selectedDate === item.dateStr ? 'selected' : ''}`}
                      onClick={() => item.hasSlots && setSelectedDate(item.dateStr)}
                    >
                      <div className="d-num">{item.dayNum}</div>
                      <div className="d-name">{item.dayName}</div>
                    </div>
                  ))}
                </div>

                <div className="foot-nav">
                  <span></span>
                  <button type="button" className="btn btn-primary btn-sm" disabled={!selectedDate} onClick={handleNextToStep2}>Avançar</button>
                </div>
              </div>

              {/* STEP 2: HORÁRIO */}
              <div className={`step-content ${step === 2 ? 'active' : ''}`}>
                <h2 style={{ fontSize: '18px', marginBottom: '4px' }}>Selecione o Horário</h2>
                <p style={{ color: 'var(--ink-soft)', fontSize: '14px', margin: '0 0 20px' }}>{fmtLabel(selectedDate)}</p>
                
                <div className="time-grid">
                  {selectedDaySlots.length === 0 ? (
                    <p className="empty-note">Não há horários disponíveis para esta data. Por favor, escolha outro dia.</p>
                  ) : (
                    selectedDaySlots.map((slot) => (
                      <div
                        key={slot}
                        className={`pick-card ${selectedTime === slot ? 'selected' : ''}`}
                        onClick={() => setSelectedTime(slot)}
                      >
                        <div className="d-num" style={{ fontSize: '15px' }}>{slot}</div>
                      </div>
                    ))
                  )}
                </div>

                <div className="foot-nav">
                  <button type="button" className="btn btn-ghost btn-sm" onClick={() => setStep(1)}>Voltar</button>
                  <button type="button" className="btn btn-primary btn-sm" disabled={!selectedTime} onClick={() => setStep(3)}>Avançar</button>
                </div>
              </div>

              {/* STEP 3: DADOS */}
              <div className={`step-content ${step === 3 ? 'active' : ''}`}>
                <h2 style={{ fontSize: '18px', marginBottom: '16px' }}>Dados da Consulta</h2>
                
                <label className="field-label" style={{ marginBottom: '10px' }}>Tipo de Atendimento / Motivo</label>
                <div className="mission-grid">
                  {[
                    { value: 'Limpeza', title: 'Check-up e Limpeza', desc: 'Prevenção e higienização', icon: <path d="M12 22s8-4 8-10V5l-8-3-8 3v7c0 6 8 10 8 10z"/> },
                    { value: 'Primeira consulta', title: 'Primeira Consulta', desc: 'Avaliação diagnóstica inicial', icon: <path d="M16 7a4 4 0 11-8 0 4 4 0 018 0zM12 14a7 7 0 00-7 7h14a7 7 0 00-7-7z"/> },
                    { value: 'Dente doendo', title: 'Sintomatologia / Dor', desc: 'Avaliação de urgência', icon: <path d="M12 9v2m0 4h.01m-6.938 4h13.856c1.54 0 2.502-1.667 1.732-3L13.732 4c-.77-1.333-2.694-1.333-3.464 0L3.34 16c-.77 1.333.192 3 1.732 3z"/> },
                    { value: 'Aparelho', title: 'Ortodontia', desc: 'Manutenção ou avaliação', icon: <><path d="M10.325 4.317c.426-1.756 2.924-1.756 3.35 0a1.724 1.724 0 002.573 1.066c1.543-.94 3.31.826 2.37 2.37a1.724 1.724 0 001.065 2.572c1.756.426 1.756 2.924 0 3.35a1.724 1.724 0 00-1.066 2.573c.94 1.543-.826 3.31-2.37 2.37a1.724 1.724 0 00-2.572 1.065c-.426 1.756-2.924 1.756-3.35 0a1.724 1.724 0 00-2.573-1.066c-1.543.94-3.31-.826-2.37-2.37a1.724 1.724 0 00-1.065-2.572c-1.756-.426-1.756-2.924 0-3.35a1.724 1.724 0 001.066-2.573c-.94-1.543.826-3.31 2.37-2.37.996.608 2.296.07 2.572-1.065z"/><path d="M15 12a3 3 0 11-6 0 3 3 0 016 0z"/></> }
                  ].map((m) => (
                    <label key={m.value} className={`mission-card ${selectedMission === m.value ? 'selected' : ''}`} onClick={() => setSelectedMission(m.value)}>
                      <input type="radio" name="mission" value={m.value} checked={selectedMission === m.value} readOnly style={{ display: 'none' }} />
                      <div className="icon-box"><svg width="20" height="20" fill="none" stroke="currentColor" strokeWidth="2" viewBox="0 0 24 24">{m.icon}</svg></div>
                      <div><b>{m.title}</b><span>{m.desc}</span></div>
                    </label>
                  ))}
                </div>

                <div className="form-grid" style={{ marginTop: '20px' }}>
                  <div>
                    <label className="field-label">Nome do paciente</label>
                    <input type="text" placeholder="Ex: Lucas Gabriel" value={formData.childName} onChange={(e) => setFormData({ ...formData, childName: e.target.value })} />
                  </div>
                  <div>
                    <label className="field-label">Idade do paciente</label>
                    <input type="number" min="0" max="120" placeholder="Ex: 25" value={formData.childAge} onChange={(e) => setFormData({ ...formData, childAge: e.target.value })} />
                  </div>
                  
                  {showParentField && (
                    <div style={{ gridColumn: '1 / -1' }}>
                      <label className="field-label">Nome do responsável <span style={{ fontWeight: 400, color: 'var(--ink-soft)' }}>(Obrigatório para menores de 18 anos)</span></label>
                      <input type="text" placeholder="Ex: Ana Souza" value={formData.parentName} onChange={(e) => setFormData({ ...formData, parentName: e.target.value })} />
                    </div>
                  )}

                  <div style={{ gridColumn: '1 / -1' }}>
                    <label className="field-label">WhatsApp / Telefone de Contato</label>
                    <input type="tel" placeholder="(11) 90000-0000" value={formData.phone} onChange={(e) => setFormData({ ...formData, phone: e.target.value })} />
                  </div>
                </div>
                
                {step3Error && <p style={{ color: 'var(--danger)', fontSize: '13px', marginTop: '12px' }}>Por favor, preencha todos os campos obrigatórios para continuar.</p>}

                <div className="foot-nav">
                  <button type="button" className="btn btn-ghost btn-sm" onClick={() => setStep(2)}>Voltar</button>
                  <button type="button" className="btn btn-primary btn-sm" onClick={handleNextToStep4}>Revisar Agendamento</button>
                </div>
              </div>

              {/* STEP 4: REVISÃO */}
              <div className={`step-content ${step === 4 ? 'active' : ''}`}>
                <h2 style={{ fontSize: '18px', marginBottom: '16px' }}>Confirmação dos Dados</h2>
                
                <div className="summary-box-ui">
                  <div className="summary-row"><span>Data</span><b>{fmtLabel(selectedDate)}</b></div>
                  <div className="summary-row"><span>Horário</span><b>{selectedTime}</b></div>
                  <div className="summary-row"><span>Procedimento</span><b>{selectedMission}</b></div>
                  <div className="summary-row"><span>Paciente</span><b>{formData.childName}, {formData.childAge} anos</b></div>
                  {showParentField && <div className="summary-row"><span>Responsável Legal</span><b>{formData.parentName}</b></div>}
                  <div className="summary-row"><span>Telefone de Contato</span><b>{formData.phone}</b></div>
                </div>

                <div className="foot-nav">
                  <button type="button" className="btn btn-ghost btn-sm" onClick={() => setStep(3)}>Voltar</button>
                  <button type="button" className="btn btn-primary btn-sm" disabled={isSubmitting} onClick={handleConfirmBooking}>
                    {isSubmitting ? "Processando..." : "Confirmar Agendamento"}
                  </button>
                </div>
              </div>

              {/* STEP 5: SUCESSO */}
              <div className={`step-content ${step === 5 ? 'active' : ''}`}>
                <div className="success-wrap">
                  <svg className="success-icon" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2"><path d="M22 11.08V12a10 10 0 1 1-5.93-9.14"/><polyline points="22 4 12 14.01 9 11.01"/></svg>
                  <h2 style={{ fontSize: '22px' }}>Consulta Agendada!</h2>
                  <p style={{ color: 'var(--ink-soft)', maxWidth: '40ch', margin: '8px auto 0', fontSize: '14px' }}>O agendamento foi registrado com sucesso. Apresente o código abaixo na recepção.</p>
                  
                  {createdBookingDetails && (
                    <div style={{ textAlign: 'left', maxWidth: '380px', margin: '20px auto 0' }}>
                      <div style={{ background: 'var(--primary-light)', border: '1px dashed var(--primary)', borderRadius: '8px', padding: '14px', textAlign: 'center', marginBottom: '16px' }}>
                        <div style={{ fontSize: '11px', color: 'var(--primary)', fontWeight: 700, letterSpacing: '.5px' }}>CÓDIGO DE AGENDAMENTO</div>
                        <div style={{ fontSize: '20px', fontWeight: 700, color: 'var(--primary)', letterSpacing: '1px', marginTop: '2px' }}>{createdBookingCode}</div>
                      </div>
                      <div className="summary-row"><span>Data</span><b>{fmtLabel(selectedDate)}</b></div>
                      <div className="summary-row"><span>Horário</span><b>{selectedTime}</b></div>
                      <div className="summary-row"><span>Paciente</span><b>{createdBookingDetails.childName}</b></div>
                    </div>
                  )}

                  <button type="button" className="btn btn-primary" style={{ marginTop: '24px' }} onClick={handleResetBooking}>Realizar Novo Agendamento</button>
                </div>
              </div>

            </form>
          </div>
        </div>
      </section>

      {/* FOOTER */}
      <footer>
        <div className="container wrap">
          <span>Sorriso Mágico · Odontologia Integrada · (11) 4000-0000</span>
          <button className="link-btn" onClick={() => setIsAdminOpen(true)}>Acesso Restrito Equipe</button>
        </div>
      </footer>

      {/* MODAL ADMIN */}
      <div className={`modal-backdrop ${isAdminOpen ? 'active' : ''}`}>
        <div className="modal">
          {!isAdminAuthenticated ? (
            <div>
              <span className="admin-tag">Área Restrita</span>
              <h2>Acesso Administrativo</h2>
              <p style={{ color: 'var(--ink-soft)', fontSize: '14px' }}>Informe a senha de acesso para gerenciar consultas.</p>
              <input type="password" placeholder="Senha de acesso" style={{ marginTop: '12px' }} value={adminPass} onChange={(e) => setAdminPass(e.target.value)} />
              {adminError && <p style={{ color: 'var(--danger)', fontSize: '13px', marginTop: '8px' }}>Senha incorreta.</p>}
              <div style={{ display: 'flex', justifyContent: 'flex-end', gap: '10px', marginTop: '20px' }}>
                <button className="btn btn-ghost btn-sm" onClick={() => setIsAdminOpen(false)}>Cancelar</button>
                <button className="btn btn-primary btn-sm" disabled={adminLoading} onClick={handleAdminEnter}>
                  {adminLoading ? "Verificando..." : "Acessar Painel"}
                </button>
              </div>
            </div>
          ) : (
            <div>
              <span className="admin-tag">Painel de Controle</span>
              <h2>Gestão de Agendamentos</h2>
              
              <div>
                {adminRows.length === 0 ? (
                  <p className="empty-note">Nenhum agendamento registrado até o momento.</p>
                ) : (
                  <table>
                    <thead>
                      <tr>
                        <th>Código</th>
                        <th>Data</th>
                        <th>Hora</th>
                        <th>Paciente</th>
                        <th>Responsável</th>
                        <th>Contato</th>
                        <th>Ação</th>
                      </tr>
                    </thead>
                    <tbody>
                      {adminRows.map((r) => (
                        <tr key={`${r.date}-${r.idx}`}>
                          <td style={{ fontWeight: 700, color: 'var(--primary)' }}>{makeBookingCode(r.childName, r.date, r.time)}</td>
                          <td>{fmtLabel(r.date)}</td>
                          <td>{r.time}</td>
                          <td>{r.childName} ({r.childAge})</td>
                          <td>{r.parentName}</td>
                          <td>{r.phone}</td>
                          <td><button className="tag-danger" onClick={() => handleCancelBooking(r.date, r.idx)}>Cancelar</button></td>
                        </tr>
                      ))}
                    </tbody>
                  </table>
                )}
              </div>

              <h2 style={{ marginTop: '28px', fontSize: '16px' }}>Bloqueio de Calendário</h2>
              <p style={{ color: 'var(--ink-soft)', fontSize: '13px' }}>Inative datas para feriados ou recessos.</p>
              <div className="admin-block-row">
                <input type="date" style={{ maxWidth: '180px' }} value={blockDayInput} onChange={(e) => setBlockDayInput(e.target.value)} />
                <button className="btn btn-primary btn-sm" onClick={handleBlockDay}>Bloquear Data</button>
              </div>

              <div style={{ marginTop: '12px' }}>
                {blockedDays.length === 0 ? (
                  <p className="empty-note" style={{ padding: '8px 0' }}>Nenhuma data bloqueada.</p>
                ) : (
                  [...blockedDays].sort().map((d) => (
                    <div key={d} className="summary-row">
                      <span>{fmtLabel(d)}</span>
                      <button className="tag-danger" onClick={() => handleUnblockDay(d)}>Desbloquear</button>
                    </div>
                  ))
                )}
              </div>

              <div style={{ display: 'flex', justifyContent: 'flex-end', marginTop: '24px' }}>
                <button className="btn btn-ghost btn-sm" onClick={() => {
                  setIsAdminOpen(false);
                  setIsAdminAuthenticated(false);
                  setAdminPass('');
                }}>Fechar Painel</button>
              </div>
            </div>
          )}
        </div>
      </div>
    </>
  );
